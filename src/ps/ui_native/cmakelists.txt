cmake_minimum_required(VERSION 3.18)
project(board_light_overlay C)
set(CMAKE_C_STANDARD 11)
set(CMAKE_POSITION_INDEPENDENT_CODE ON)
set(CMAKE_C_STANDARD_REQUIRED ON)
if(NOT LVGL_SOURCE_DIR OR NOT EXISTS "${LVGL_SOURCE_DIR}/CMakeLists.txt")
  message(FATAL_ERROR "Pass -DLVGL_SOURCE_DIR=<local LVGL v9.6.0 source>; no network fetch is performed")
endif()
file(READ "${LVGL_SOURCE_DIR}/include/lvgl/lv_version.h" LVGL_VERSION_HEADER)
foreach(PART MAJOR MINOR PATCH)
  if(PART STREQUAL "MAJOR")
    set(EXPECTED 9)
  elseif(PART STREQUAL "MINOR")
    set(EXPECTED 6)
  else()
    set(EXPECTED 0)
  endif()
  if(NOT LVGL_VERSION_HEADER MATCHES "#define LVGL_VERSION_${PART} ${EXPECTED}([\r\n]|$)")
    message(FATAL_ERROR "This prototype requires LVGL 9.6.0")
  endif()
endforeach()
add_compile_definitions(LV_CONF_INCLUDE_SIMPLE)
include_directories("${CMAKE_CURRENT_SOURCE_DIR}")
set(CONFIG_LV_BUILD_EXAMPLES OFF CACHE BOOL "" FORCE)
set(CONFIG_LV_BUILD_DEMOS OFF CACHE BOOL "" FORCE)
set(CONFIG_LV_USE_THORVG OFF CACHE BOOL "" FORCE)
set(CONFIG_LV_USE_THORVG_INTERNAL OFF CACHE BOOL "" FORCE)
set(LV_BUILD_TESTS OFF CACHE BOOL "" FORCE)
add_subdirectory("${LVGL_SOURCE_DIR}" "${CMAKE_CURRENT_BINARY_DIR}/lvgl")
add_library(board_ps_ui SHARED archive_bridge.c rgb565_blend.c fonts/board_ui_cjk24.c)
target_link_libraries(board_ps_ui PRIVATE lvgl)
target_compile_options(board_ps_ui PRIVATE -Wall -Wextra)
